// Supabase Edge Function: fiba-proxy // Descarga server-side el feed de FIBA LiveStats y lo devuelve con cabeceras CORS, // para que tu app (quintetos.html / ScoutPRO) pueda leerlo desde el navegador. // // Deploy (una sola vez): // supabase functions deploy fiba-proxy --no-verify-jwt // // Uso desde el navegador: // GET https://TU-PROYECTO.supabase.co/functions/v1/fiba-proxy?matchId=2853460 // (también acepta POST {"matchId":"2853460"}) // // El parámetro --no-verify-jwt lo hace público (sin token). Si preferís protegerlo, // quitá ese flag y mandá el header Authorization: Bearer desde la app. import { serve } from "https://deno.land/std@0.224.0/http/server.ts"; const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", }; function json(body: unknown, status = 200): Response { return new Response(JSON.stringify(body), { status, headers: { ...CORS, "Content-Type": "application/json; charset=utf-8" }, }); } serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: CORS }); try { // matchId puede venir por querystring (?matchId=) o en el body JSON const url = new URL(req.url); let matchId = url.searchParams.get("matchId") ?? ""; if (!matchId && req.method === "POST") { const body = await req.json().catch(() => ({})); matchId = String(body?.matchId ?? ""); } // también acepta que le manden la URL/pbp.html entera y le saca el número const m = matchId.match(/(\d{5,})/); matchId = m ? m[1] : ""; if (!matchId) return json({ error: "Falta matchId (el número del partido)." }, 400); const feed = `https://fibalivestats.dcd.shared.geniussports.com/data/${matchId}/data.json`; const upstream = await fetch(feed, { headers: { // algunos edges de Genius responden mejor con un UA de navegador "User-Agent": "Mozilla/5.0 (compatible; ScoutPRO/1.0)", "Accept": "application/json, text/plain, */*", }, }); if (!upstream.ok) { return json( { error: `El feed respondió ${upstream.status}.`, matchId, feed }, 502, ); } const text = await upstream.text(); // lo reenviamos tal cual, con CORS return new Response(text, { headers: { ...CORS, "Content-Type": "application/json; charset=utf-8", "Cache-Control": "public, max-age=20", // el partido en vivo cambia; 20s está bien }, }); } catch (e) { return json({ error: "Error en el proxy: " + String(e) }, 500); } });
